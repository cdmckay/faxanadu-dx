cmake_minimum_required(VERSION 3.20)
project(FaxanaduSlainTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_slain.c is included by the test, against a model of the machine.
add_executable(slain_test slain_test.c)
target_include_directories(slain_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(slain_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME slain_unit COMMAND slain_test)
