cmake_minimum_required(VERSION 3.20)
project(FaxanaduSpellCostTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_spell_cost.c is included by the test, against a model of the machine.
add_executable(spell_cost_test spell_cost_test.c)
target_include_directories(spell_cost_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(spell_cost_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME spell_cost_unit COMMAND spell_cost_test)
