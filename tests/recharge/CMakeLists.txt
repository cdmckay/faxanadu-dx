cmake_minimum_required(VERSION 3.20)
project(FaxanaduRechargeTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_recharge.c is included by the test, against a model of the machine.
add_executable(recharge_test recharge_test.c)
target_include_directories(recharge_test PRIVATE ../.. ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(recharge_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME recharge_unit COMMAND recharge_test)
