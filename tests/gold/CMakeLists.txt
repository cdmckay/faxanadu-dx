cmake_minimum_required(VERSION 3.20)
project(FaxanaduGoldTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_gold.c is included by the test, against a model of the machine.
add_executable(gold_test gold_test.c)
target_include_directories(gold_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(gold_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME gold_unit COMMAND gold_test)
