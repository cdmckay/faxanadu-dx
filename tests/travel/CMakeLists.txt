cmake_minimum_required(VERSION 3.20)
project(FaxanaduTravelTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_travel.c is included by the test, against a model of the machine.
add_executable(travel_test travel_test.c)
target_include_directories(travel_test PRIVATE ../.. ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(travel_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME travel_unit COMMAND travel_test)
