cmake_minimum_required(VERSION 3.20)
project(FaxanaduUseItemTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_use_item.c is included by the test, against a model of the machine.
add_executable(use_item_test use_item_test.c)
target_include_directories(use_item_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(use_item_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME use_item_unit COMMAND use_item_test)
