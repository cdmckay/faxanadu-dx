cmake_minimum_required(VERSION 3.20)
project(FaxanaduFlickerTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_flicker.c is included by the test, against a model of the game's sprite slots.
add_executable(flicker_test flicker_test.c)
target_include_directories(flicker_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(flicker_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME flicker_unit COMMAND flicker_test)
