cmake_minimum_required(VERSION 3.20)
project(FaxanaduItemBoxTests C)
set(CMAKE_C_STANDARD 11)
if(NOT NESRECOMP_ROOT)
    message(FATAL_ERROR "Set NESRECOMP_ROOT to the game's cycle engine checkout")
endif()
# fax_item_box.c is included by the test, against a model of the machine.
add_executable(item_box_test item_box_test.c)
target_include_directories(item_box_test PRIVATE ../../src ${NESRECOMP_ROOT}/runner/include ${NESRECOMP_ROOT}/runner/cyc)
if(NOT MSVC)
    target_compile_options(item_box_test PRIVATE -Wall -Wextra)
endif()
enable_testing()
add_test(NAME item_box_unit COMMAND item_box_test)
